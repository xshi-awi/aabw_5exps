\RequirePackage{color} %DIF PREAMBLE

\providecommand{\DIFdeltex}[1]{{\protect\color{red}\sout{#1}}} %DIF PREAMBLE
%DIF The two flags tell \rfig and \docxcomment whether they sit inside a deleted or an added block.
\newif\ifDIFindel %DIF PREAMBLE
\newif\ifDIFinadd %DIF PREAMBLE
\DeclareRobustCommand{\DIFaddbegin}{\global\DIFinaddtrue} %DIF PREAMBLE
\DeclareRobustCommand{\DIFaddend}{\global\DIFinaddfalse} %DIF PREAMBLE
\DeclareRobustCommand{\DIFdelbegin}{\global\DIFindeltrue} %DIF PREAMBLE
\DeclareRobustCommand{\DIFdelend}{\global\DIFindelfalse} %DIF PREAMBLE
\providecommand{\DIFmodbegin}{} %DIF PREAMBLE
\providecommand{\DIFmodend}{} %DIF PREAMBLE
\providecommand{\DIFaddFL}[1]{\DIFadd{#1}} %DIF PREAMBLE
\providecommand{\DIFdelFL}[1]{\DIFdel{#1}} %DIF PREAMBLE
\DeclareRobustCommand{\DIFaddbeginFL}{\global\DIFinaddtrue} %DIF PREAMBLE
\DeclareRobustCommand{\DIFaddendFL}{\global\DIFinaddfalse} %DIF PREAMBLE
\DeclareRobustCommand{\DIFdelbeginFL}{\global\DIFindeltrue} %DIF PREAMBLE
\DeclareRobustCommand{\DIFdelendFL}{\global\DIFindelfalse} %DIF PREAMBLE
\providecommand{\DIFadd}[1]{\texorpdfstring{\DIFaddtex{#1}}{#1}} %DIF PREAMBLE
\providecommand{\DIFdel}[1]{\texorpdfstring{\DIFdeltex{#1}}{}} %DIF PREAMBLE
\makeatletter %DIF PREAMBLE
\let\sout@orig\sout %DIF PREAMBLE
\renewcommand{\sout}[1]{\ifmmode\text{\sout@orig{\ensuremath{#1}}}\else\sout@orig{#1}\fi} %DIF PREAMBLE
\makeatother %DIF PREAMBLE
%DIF A deleted figure is shown at half width in a red frame, labelled, and does not take a figure number,
%DIF so the numbers R1, R2, ... in the tracked version are those of the new letter. An added figure gets a blue frame.
\renewcommand{\rfig}[3]{\ifDIFindel\par\medskip\noindent\begin{minipage}{\linewidth}\centering{\setlength{\fboxsep}{1pt}\fcolorbox{red}{white}{\includegraphics[width=0.5\linewidth,height=0.3\textheight,keepaspectratio]{figures/#2}}}\par\smallskip{\small\color{red}\textbf{Deleted figure.}}\ {\small #3}\end{minipage}\par\medskip\else\ifDIFinadd\par\medskip\noindent\begin{minipage}{\linewidth}\centering{\setlength{\fboxsep}{1pt}\fcolorbox{blue}{white}{\includegraphics[width=#1\linewidth,height=0.6\textheight,keepaspectratio]{figures/#2}}}\captionof{figure}{#3}\end{minipage}\par\medskip\else\par\medskip\noindent\begin{minipage}{\linewidth}\centering\includegraphics[width=#1\linewidth,height=0.6\textheight,keepaspectratio]{figures/#2}\captionof{figure}{#3}\end{minipage}\par\medskip\fi\fi} %DIF PREAMBLE
%DIF A deleted Word comment is struck through as a whole, not only its text.
